\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{arrows.meta,positioning,calc,fit,shapes.geometric}

% ---- a tiny SysML v2 vocabulary, built from scratch on plain TikZ ----
\tikzset{
  sysml block/.style={
    draw, rectangle, align=left, inner sep=0pt,
    text width=46mm, minimum height=9mm,
    font=\footnotesize\sffamily,
    append after command={
      \pgfextra{\pgfmathsetlengthmacro\blkh{\pgfkeysvalueof{/pgf/minimum height}}}
    }
  },
  sysml head/.style={font=\footnotesize\sffamily\bfseries},
  port/.style={draw, rectangle, fill=white, minimum size=3mm, inner sep=0pt},
  comp/.style={-{Diamond[length=4mm,width=2.6mm,fill=black]}},
  gen/.style={-{Triangle[length=3.6mm,width=3.2mm,open,fill=white]}},
  flow/.style={-{Stealth[length=2.6mm]}, dashed},
}
% block macro: #1 name, #2 compartment body (may be empty)
\newcommand{\block}[3][0,0]{%
  \node[sysml block, label={[sysml head, anchor=north west, inner sep=2pt]north west:{«block» #2}}]
    (#2) at #1 {#3\vphantom{Ag}};}

\begin{document}
\begin{tikzpicture}[node distance=14mm and 26mm]

  % --- blocks ---------------------------------------------------------
  \node[sysml block] (veh) at (0,0)
    {\textbf{«block»} Vehicle\\[1pt]\hrule\vspace{2pt}
     \textcolor{gray}{«part»}\\
     engine: Engine [1]\\
     wheels: Wheel [4]\\[1pt]\hrule\vspace{2pt}
     \textcolor{gray}{«attribute»}\\
     mass: MassValue [1]};

  \node[sysml block, right=of veh] (eng)
    {\textbf{«block»} Engine\\[1pt]\hrule\vspace{2pt}
     \textcolor{gray}{«attribute»}\\
     power: PowerValue [1]};

  \node[sysml block, below=of eng] (whl)
    {\textbf{«block»} Wheel\\[1pt]\hrule\vspace{2pt}
     \textcolor{gray}{«attribute»}\\
     diameter: LengthValue [1]};

  % --- ports on the border --------------------------------------------
  \node[port] (veh_p1) at ($(veh.west)+(0,3mm)$) {};
  \node[port] (veh_p2) at ($(veh.east)+(0,3mm)$) {};
  \node[port, label=right:{\scriptsize fuelIn}] (eng_p1) at (eng.west) {};

  % --- relations -------------------------------------------------------
  \draw[comp] (eng.north -| eng.north) -- ++(0,5mm) -| node[pos=0.25, above, font=\scriptsize]{1} (veh.south);
  \draw[comp] (whl) -- node[above, font=\scriptsize, pos=0.35]{4} (veh.south west);
  \draw[flow] (veh_p2) -- node[above, font=\scriptsize]{fuel} (eng_p1);

  % --- requirement -----------------------------------------------------
  \node[sysml block, below=of veh, text width=52mm] (req)
    {\textbf{«requirement»} massLimit\\[1pt]\hrule\vspace{2pt}
     \textcolor{gray}{«attribute»}\\
     id: "REQ-014"\\
     text: "Vehicle mass shall not exceed 1800 kg."};

  \draw[-{Stealth[length=2.6mm]}, dotted]
    (req.north) -- node[right, font=\scriptsize]{«satisfy»} (veh.south west);

\end{tikzpicture}
\end{document}
