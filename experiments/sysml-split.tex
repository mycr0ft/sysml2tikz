\tikzset{

  gen/.style={-{Triangle[length=3.2mm,width=3mm,open,fill=white]}, thick},
  dep/.style={-{Stealth[length=2.4mm]}, densely dotted},
  flow/.style={-{Stealth[length=2.4mm]}, dashed},
}
